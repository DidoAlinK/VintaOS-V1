# PyInstaller version resource for the Vinta OS executable.
#
# Without this the .exe has no embedded metadata: Windows shows a blank
# "Details" tab, Add/Remove Programs shows no publisher, and the file looks
# anonymous to both a cautious IT administrator and an antivirus heuristic.
# Keep the numbers in step with APP_VERSION in launcher_os.py.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 6, 3, 0),
    prodvers=(1, 6, 3, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
    ),
  kids=[
    StringFileInfo([
      StringTable(
        u'040904B0',
        [StringStruct(u'CompanyName', u'Vinta Automation'),
         StringStruct(u'FileDescription', u'Vinta OS — school management system'),
         StringStruct(u'FileVersion', u'1.6.3.0'),
         StringStruct(u'InternalName', u'Vinta OS'),
         StringStruct(u'LegalCopyright', u'Copyright (C) 2026 Vinta Automation'),
         StringStruct(u'OriginalFilename', u'Vinta OS.exe'),
         StringStruct(u'ProductName', u'Vinta OS'),
         StringStruct(u'ProductVersion', u'1.6.3.0')])
      ]),
    VarFileInfo([VarStruct(u'Translation', [1033, 1200])])
  ]
)
